## 2. 用户数据管理接口 (UserData API)

### 2.1 核心UserData接口

#### 2.1.1 获取用户数据文件
```typescript
// 接口路径: GET /api/userdata/{filename}
// 功能: 获取指定的用户数据文件内容
// 前端调用: api.getUserData(filename)
// 示例: api.getUserData('comfy.settings.json')

async getUserData(file: string, options?: RequestInit): Promise<Response> {
  return this.fetchApi(`/userdata/${encodeURIComponent(file)}`, options);
}
```

#### 2.1.2 保存用户数据文件
```typescript
// 接口路径: POST /api/userdata/{filename}
// 功能: 保存用户数据文件
// 前端调用: api.storeUserData(filename, data, options)
// 示例: api.storeUserData('comfy.settings.json', settingsData)

async storeUserData(
  file: string,
  data: any,
  options: {
    overwrite?: boolean;      // 是否覆盖现有文件，默认true
    stringify?: boolean;      // 是否JSON序列化，默认true
    throwOnError?: boolean;   // 是否抛出错误，默认true
    full_info?: boolean;      // 是否返回完整信息，默认false
  } = {}
): Promise<Response>
```

### 3.1 获取可用模型列表
```typescript
// 通过object_info接口获取各类型模型列表
// 前端调用: api.getAvailableModels()

async getAvailableModels(): Promise<{
  checkpoints: string[];           // 主模型列表
  vae: string[];                  // VAE模型列表
  loras: string[];                // LoRA模型列表
  controlnet: string[];           // ControlNet模型列表
  upscale_models: string[];       // 超分模型列表
  embeddings: string[];           // 嵌入模型列表

  const data = await response.json();
  
  return {
    checkpoints: data.CheckpointLoaderSimple?.input?.required?.ckpt_name?.[0] || [],
    vae: data.VAELoader?.input?.required?.vae_name?.[0] || [],
    loras: data.LoraLoader?.input?.required?.lora_name?.[0] || [],
    controlnet: data.ControlNetLoader?.input?.required?.control_net_name?.[0] || [],
    upscale_models: data.UpscaleModelLoader?.input?.required?.model_name?.[0] || [],
    embeddings: data.TextualInversionLoader?.input?.required?.embedding_name?.[0] || []
  };
}
```

```typescript
// 核心模板JSON文件获取

// 模板工作流JSON格式 (ComfyWorkflowJSON)
interface ComfyWorkflowJSON {
  last_node_id: number;           // 最后一个节点ID
  last_link_id: number;           // 最后一个连接ID

    pos: [number, number];        // 节点位置 [x, y]
    size: [number, number];       // 节点尺寸 [width, height]

    order: number;                // 执行顺序

    inputs?: Array<{              // 输入连接
      name: string;               // 输入名称
      type: string;               // 输入类型
      link: number | null;        // 连接ID
    }>;
    outputs?: Array<{             // 输出连接
      name: string;               // 输出名称
      type: string;               // 输出类型
      links: number[] | null;     // 连接的链接ID列表

    properties: Record<string, any>; // 节点属性
    widgets_values?: any[];       // 控件值
  }>;
  links: Array<[                  // 连接列表
    number,                       // 连接ID

    number,                       // 源输出索引

  ]>;
  groups: Array<{                 // 节点组

    bounding: [number, number, number, number]; // 边界 [x, y, width, height]
    color: string;                // 组颜色
```

```typescript
// 核心模板缩略图
// 接口路径: GET /templates/{templateName}.{mediaSubtype}
// 功能: 获取核心模板的预览缩略图
// 前端调用: api.getTemplateThumbnailUrl(template, 'default')

// 自定义节点模板缩略图
// 接口路径: GET /api/workflow_templates/{moduleName}/{templateName}.{mediaSubtype}  
// 功能: 获取自定义节点模板的预览缩略图
// 前端调用: api.getTemplateThumbnailUrl(template, moduleName)

// 缩略图URL生成方法
getTemplateThumbnailUrl(
  template: TemplateInfo, 
  sourceModule: string, 
  index = ''
): string {
  const basePath = sourceModule === 'default'
    ? api.fileURL(`/templates/${template.name}`)
    : api.apiURL(`/workflow_templates/${sourceModule}/${template.name}`);
    
  return `${basePath}.${template.mediaSubtype}`;
}
```

### 4.2 保存工作流
```typescript
// 使用UserData API保存工作流
// 前端调用: api.saveWorkflow(filename, workflowData)

async saveWorkflow(filename: string, workflow: WorkflowData): Promise<void> {
  // 保存工作流文件
  await this.storeUserData(`workflows/${filename}.json`, workflow);
  
  // 更新工作流索引
  const index = await this.getWorkflowIndex();
  index.workflows[filename] = {
    name: workflow.extra?.title || filename,
    description: workflow.extra?.description,
    created: Date.now(),
    modified: Date.now(),
    tags: workflow.extra?.tags || [],
    category: workflow.extra?.category
  };
  await this.storeUserData('workflows/.index.json', index);
}
```

## 5. 任务队列接口

### 5.1 提交任务到队列
```typescript
// 接口路径: POST /prompt
// 功能: 提交工作流到执行队列
// 前端调用: api.queuePrompt(workflow, clientId)

async queuePrompt(workflow: WorkflowData, clientId?: string): Promise<{
  prompt_id: string;              // 任务ID
  number: number;                 // 队列中的位置
  node_errors: Record<string, any>; // 节点错误信息
}> {
  const body = {
    prompt: workflow,
    client_id: clientId || this.clientId
  };
  
  const response = await this.fetchApi('/prompt', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body)
  });
  
  return response.json();
}
```

## 🛠️ 前端API调用示例

### 示例1: 加载默认配置
```typescript
// 在Vue组件中加载默认配置
import { ref, onMounted } from 'vue'
import { api } from '@/scripts/api'

export default {
  setup() {
    const settings = ref(null)
    const templates = ref(null)
    
    onMounted(async () => {
      try {
        // 加载用户设置
        const settingsResponse = await api.getUserData('comfy.settings.json')
        settings.value = await settingsResponse.json()
        
        // 加载工作流模板
        const templatesResponse = await api.getUserData('comfy.templates.json')
        templates.value = await templatesResponse.json()
        
        console.log('配置加载完成:', {
          settings: settings.value,
          templates: templates.value
        })
        
      } catch (error) {
        console.error('加载配置失败:', error)
        // 使用默认配置
        settings.value = getDefaultSettings()
      }
    })
    
    return {
      settings,
      templates
    }
  }
}
```
